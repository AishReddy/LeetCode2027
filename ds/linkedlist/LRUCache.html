LeetCode #146

LRU Cache

O(1) get & put using a HashMap + Doubly Linked List. Java implementation with full annotations.

Difficulty Medium
get() Time O(1)
put() Time O(1)
Space O(n)
Pattern HashMap + DLL
LRUCache.java
1// ─── LRU Cache — Custom Doubly Linked List + HashMap ─────────────────────
2// Time: O(1) for get/put | Space: O(capacity)
3
4import java.util.HashMap;
5
6class LRUCache {
7
8 // ── Inner Node class ─────────────────────────────────────────────────
9 private class Node {
10 int key, val;
11 Node prev, next;
12
13 Node(int key, int val) {
14 this.key = key;
15 this.val = val;
16 }
17 }
18
19 // ── Fields ────────────────────────────────────────────────────────────
20 private final int capacity;
21 private final HashMap<Integer, Node> map;
22
23 // Dummy head (MRU side) and tail (LRU side)
24 private final Node head = new Node(0, 0); // most recent
25 private final Node tail = new Node(0, 0); // least recent
26
27 // ── Constructor ───────────────────────────────────────────────────────
28 public LRUCache(int capacity) {
29 this.capacity = capacity;
30 map = new HashMap<>();
31
32 // Wire the two dummy sentinels together
33 head.next = tail;
34 tail.prev = head;
35 }
36
37 // ── get ───────────────────────────────────────────────────────────────
38 @Override
39 public int get(int key) {
40 if (!map.containsKey(key)) return -1;
41
42 Node node = map.get(key);
43 remove(node); // unlink from current position
44 insertFront(node); // re-insert at MRU position
45 return node.val;
46 }
47
48 // ── put ───────────────────────────────────────────────────────────────
49 public void put(int key, int value) {
50 if (map.containsKey(key)) {
51 remove(map.get(key)); // remove stale node
52 }
53
54 Node node = new Node(key, value);
55 map.put(key, node);
56 insertFront(node);
57
58 if (map.size() > capacity) {
59 Node lru = tail.prev; // node just before dummy tail
60 remove(lru);
61 map.remove(lru.key);
62 }
63 }
64
65 // ── Helpers ───────────────────────────────────────────────────────────
66
67 /** Unlinks a node from wherever it is in the list. */
68 private void remove(Node node) {
69 node.prev.next = node.next;
70 node.next.prev = node.prev;
71 }
72
73 /** Inserts a node right after the dummy head (MRU position). */
74 private void insertFront(Node node) {
75 node.next = head.next;
76 node.prev = head;
77 head.next.prev = node;
78 head.next = node;
79 }
80}
81
82// ─── Example usage ───────────────────────────────────────────────────────────
83// LRUCache cache = new LRUCache(2);
84// cache.put(1, 1); // cache: {1=1}
85// cache.put(2, 2); // cache: {1=1, 2=2}
86// cache.get(1); // return 1, cache: {2=2, 1=1}
87// cache.put(3, 3); // evicts key 2, cache: {1=1, 3=3}
88// cache.get(2); // return -1 (evicted)
1// ─── LRU Cache — LinkedHashMap one-liner approach ────────────────────────
2// Leverages Java's built-in access-ordered LinkedHashMap.
3// Simpler code, same O(1) amortized complexity.
4
5import java.util.LinkedHashMap;
6import java.util.Map;
7
8class LRUCache extends LinkedHashMap<Integer, Integer> {
9
10 private final int capacity;
11
12 public LRUCache(int capacity) {
13 // accessOrder=true → iterates from least to most recently accessed
14 super(capacity, 0.75f, true);
15 this.capacity = capacity;
16 }
17
18 public int get(int key) {
19 // getOrDefault also re-orders the entry → marks it as most recent
20 return super.getOrDefault(key, -1);
21 }
22
23 public void put(int key, int value) {
24 super.put(key, value);
25 }
26
27 // Called automatically by LinkedHashMap after every put()
28 @Override
29 protected boolean removeEldestEntry(Map.Entry<Integer, Integer> eldest) {
30 // eldest = the LRU entry (head of access-ordered list)
31 return size() > capacity;
32 }
33}
34
35// ─── Note ────────────────────────────────────────────────────────────────────
36// This approach is concise but LinkedHashMap is NOT thread-safe.
37// For interviews, the custom DLL solution is preferred as it shows
38// a deeper understanding of the underlying data structure.
Time Complexity
get(key) O(1)
put(key, val) O(1)
remove (internal) O(1)
insertFront (internal) O(1)
Space Complexity
HashMap O(n)
Doubly Linked List O(n)
Total O(n)
n = capacity bounded
How it works
01 Two dummy sentinel nodes (head and tail) bracket the list, so insert/delete never needs null checks — every real node always has a valid prev and next.
02 The HashMap maps each key to its Node reference, giving O(1) lookup. The list maintains access order — head side = most recently used, tail side = least recently used.
03 get(key): Look up the node in the map. If found, call remove() then insertFront() to move it to the MRU position. Return its value (or -1 if absent).
04 put(key, val): If the key exists, remove its old node first. Create and insert a fresh node at the front. If size exceeds capacity, evict tail.prev (the LRU node) and delete it from the map.
05 LinkedHashMap shortcut: Setting accessOrder = true in the super-constructor makes Java maintain access order internally. Override removeEldestEntry() to auto-evict when over capacity.
